Create Payment Session

curl -X POST "{{ base_url }}/payments/create" \
  -H "Content-Type: application/json" \
  -H "X-Idempotency-Key: 8ecfef66-1ed4-4be0-a3a7-a5631cb1a8ab" \
  -d '{
    "public_id": "8dff4977-a180-4ef5-9a4c-47b98f5e47dd",
    "turnstile_token": "token_from_frontend"
  }'

Generate Screenshot Upload URL

curl -X POST "{{ base_url }}/payments/{payment_public_id}/upload-url" \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer {payment_token}" \
  -d '{
    "sha256": "f8f4b41d11e8f13b6d6e0ff8fd2d6d9c06fc7d03b30ef935398f4f6f9f5a6f57",
    "file_extension": "png",
    "mime_type": "image/png",
    "file_size": 302188
  }'

Mint Payment Token (Session Restore)

curl -X POST "{{ base_url }}/payments/{payment_public_id}/token" \
  -H "Content-Type: application/json" \
  -d '{
    "public_id": "8dff4977-a180-4ef5-9a4c-47b98f5e47dd",
    "session_id": "sess_abc123"
  }'

Verify Payment Screenshot

curl -X POST "{{ base_url }}/payments/{payment_public_id}/verify-upload" \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer {payment_token}" \
  -H "X-Idempotency-Key: 4ce6f1f2-f2ef-413c-ae1e-f6d2dc97dc84" \
  -d '{
    "upload_object_key": "payments/staging/{payment_public_id}/up_abc123.png",
    "sha256": "f8f4b41d11e8f13b6d6e0ff8fd2d6d9c06fc7d03b30ef935398f4f6f9f5a6f57",
    "file_extension": "png",
    "turnstile_token": "token_from_frontend"
  }'
